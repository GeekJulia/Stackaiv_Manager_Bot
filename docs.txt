| Entity              | Purpose                      |
| ------------------- | ---------------------------- |
| users               | Telegram users               |
| programs            | Academic programs            |
| program_activations | Tracks free/paid activations |
| groups              | Telegram group chats         |
| courses             | Courses under a program      |
| resource_types      | Notes, PQs, etc              |
| resources           | Uploaded files               |
| approvals           | Who approved a file          |
| payments            | Paystack records             |

Initial DB structure: 
-- USERS
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    telegram_id BIGINT UNIQUE NOT NULL,
    username TEXT,
    first_name TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- PROGRAMS
CREATE TABLE programs (
    id SERIAL PRIMARY KEY,
    level TEXT NOT NULL,
    college TEXT NOT NULL,
    name TEXT NOT NULL,
    created_by BIGINT REFERENCES users(telegram_id),
    created_at TIMESTAMP DEFAULT NOW(),
    is_active BOOLEAN DEFAULT FALSE
);

-- PROGRAM ACTIVATIONS
CREATE TABLE program_activations (
    id SERIAL PRIMARY KEY,
    program_id INT REFERENCES programs(id),
    is_paid BOOLEAN DEFAULT FALSE,
    activated_at TIMESTAMP DEFAULT NOW()
);

-- GROUP CHATS
CREATE TABLE groups (
    id SERIAL PRIMARY KEY,
    telegram_chat_id BIGINT UNIQUE NOT NULL,
    program_id INT REFERENCES programs(id),
    created_at TIMESTAMP DEFAULT NOW()
);

-- COURSES
CREATE TABLE courses (
    id SERIAL PRIMARY KEY,
    program_id INT REFERENCES programs(id),
    course_code TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT NOW()
);

-- RESOURCE TYPES
CREATE TABLE resource_types (
    id SERIAL PRIMARY KEY,
    name TEXT UNIQUE NOT NULL
);

-- COURSE ↔ RESOURCE TYPES
CREATE TABLE course_resource_types (
    course_id INT REFERENCES courses(id),
    resource_type_id INT REFERENCES resource_types(id),
    PRIMARY KEY (course_id, resource_type_id)
);

-- RESOURCES
CREATE TABLE resources (
    id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(id),
    resource_type_id INT REFERENCES resource_types(id),
    file_id TEXT NOT NULL,
    file_name TEXT,
    uploader_id BIGINT REFERENCES users(telegram_id),
    created_at TIMESTAMP DEFAULT NOW(),
    UNIQUE (file_id)
);

-- APPROVALS (GROUP FILE APPROVAL)
CREATE TABLE approvals (
    id SERIAL PRIMARY KEY,
    resource_id INT REFERENCES resources(id),
    approved_by BIGINT REFERENCES users(telegram_id),
    approved_at TIMESTAMP DEFAULT NOW()
);

-- PAYMENTS
CREATE TABLE payments (
    id SERIAL PRIMARY KEY,
    program_id INT REFERENCES programs(id),
    reference TEXT UNIQUE,
    amount INT,
    status TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);
